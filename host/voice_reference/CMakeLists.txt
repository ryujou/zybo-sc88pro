cmake_minimum_required(VERSION 3.20)
project(sc88_voice_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GEAR "${CMAKE_CURRENT_SOURCE_DIR}/../../external/gearmulator/source")
option(SC88_PARAMETER_OBSERVE "Observe PC parameter hooks in the fixed numerical probes" OFF)
if(SC88_PARAMETER_OBSERVE)
    set(OVERLAY "${CMAKE_CURRENT_SOURCE_DIR}/../../build/parameter-voice-overlay")
    set(VOICE_MAIN "${OVERLAY}/main.cpp")
    set(MULTIVOICE_MAIN "${OVERLAY}/multivoice.cpp")
else()
    set(OVERLAY "${CMAKE_CURRENT_SOURCE_DIR}/../../build/voice-overlay")
    set(VOICE_MAIN main.cpp)
    set(MULTIVOICE_MAIN multivoice_vectors.cpp)
endif()
add_compile_definitions(CHIPS_FORCE_NO_JIT=1 _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    add_compile_options(/utf-8 /permissive-)
endif()
add_subdirectory("${GEAR}/cpu/common" "upstream/common")
add_subdirectory("${GEAR}/ronaldo/custom_chips/xp" "upstream/xp")
add_subdirectory("${GEAR}/ronaldo/custom_chips/lsp" "upstream/lsp")
get_target_property(XP_SOURCES xpLib SOURCES)
list(REMOVE_ITEM XP_SOURCES xp_voice.cpp xp.cpp)
set_property(TARGET xpLib PROPERTY SOURCES ${XP_SOURCES} "${OVERLAY}/xp_voice.cpp" "${OVERLAY}/xp.cpp")
target_include_directories(xpLib PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(sc88_voice_vectors "${VOICE_MAIN}")
target_include_directories(sc88_voice_vectors PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(sc88_voice_vectors PRIVATE xpLib)
if(SC88_PARAMETER_OBSERVE)
    add_executable(sc88_parameter_irq_vectors "${OVERLAY}/irq_vectors.cpp")
    target_include_directories(sc88_parameter_irq_vectors PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(sc88_parameter_irq_vectors PRIVATE xpLib)
endif()
add_executable(sc88_multivoice_vectors "${MULTIVOICE_MAIN}")
target_include_directories(sc88_multivoice_vectors PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(sc88_multivoice_vectors PRIVATE xpLib)
add_executable(sc88_ramp_vectors ramp_vectors.cpp)
target_link_libraries(sc88_ramp_vectors PRIVATE xpLib)
add_executable(sc88_control_vectors control_vectors.cpp)
target_link_libraries(sc88_control_vectors PRIVATE xpLib)
add_executable(sc88_ramp_service_vectors ramp_service_vectors.cpp)
target_link_libraries(sc88_ramp_service_vectors PRIVATE xpLib)
add_executable(sc88_source_control_vectors source_control_vectors.cpp)
target_link_libraries(sc88_source_control_vectors PRIVATE xpLib)
add_executable(sc88_voice_trace_probe "${OVERLAY}/replay.cpp" ../trace/trace.cpp "${GEAR}/framework/baseLib/md5.cpp")
target_include_directories(sc88_voice_trace_probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" ../trace "${GEAR}/framework" "${GEAR}/ronaldo")
target_link_libraries(sc88_voice_trace_probe PRIVATE xpLib lspLib)
option(SC88_BOARD_PROBE "Build the bounded 64 Voice MIDI Trace observation probe" OFF)
if(SC88_BOARD_PROBE)
    add_executable(sc88_board_trace_probe "${CMAKE_CURRENT_SOURCE_DIR}/../../build/board-overlay/replay.cpp" ../trace/trace.cpp "${GEAR}/framework/baseLib/md5.cpp")
    target_include_directories(sc88_board_trace_probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" ../trace "${GEAR}/framework" "${GEAR}/ronaldo")
    target_link_libraries(sc88_board_trace_probe PRIVATE xpLib lspLib)
endif()
