cmake_minimum_required(VERSION 3.20)
project(xp3_voice_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(XP3_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build/hls-source" CACHE PATH "Generated numerical source (also used for coefficient fault test)")
add_executable(xp3_testbench testbench.cpp xp3_voice.cpp "${XP3_GENERATED_DIR}/xp3_core.cpp")
target_include_directories(xp3_testbench PRIVATE . "${XP3_GENERATED_DIR}")
target_compile_definitions(xp3_testbench PRIVATE XP3_NATIVE_TEST=1)
if(MSVC)
    target_compile_options(xp3_testbench PRIVATE /utf-8 /permissive-)
endif()
